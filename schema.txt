// Enum definitions
Enum Unit_Item_Status {
  available
  rented
  maintainance
}

Enum Transaction_Payment_Method {
  pending_payment
  qris
  cash
}

Enum Transaction_Status {
  pending
  complete
  cancel
}

Enum Payment_Link_Status {
  pending
  complete
  cancel
  expired
}

Enum Rented_Unit_Status {
  playing
  finished
  cancel
}

// Table definitions
Table unit_item {
  id int [pk, increment]
  title varchar
  description text
  rent_price float
  status Unit_Item_Status [default: 'available']
  created_at datetime [default: `now()`]
  updated_at datetime
}

Table fnb_item {
  id int [pk, increment]
  title varchar
  description text
  price float
  created_at datetime [default: `now()`]
  updated_at datetime
}

Table transaction {
  id int [pk, increment]
  customer_name varchar
  subtotal float
  total float
  payment_method Transaction_Payment_Method [default: 'pending_payment']
  status Transaction_Status [default: 'pending']
  created_at datetime [default: `now()`]
  updated_at datetime [default: `now()`]
}

Table transaction_item_unit {
  id int [pk, increment]
  transaction_id int
  unit_item_id int
  play_time float
  sub_total float
  start_time datetime [default: `now()`]
  end_time datetime
  status Rented_Unit_Status [default: 'playing']
}

Table transaction_item_fnb {
  id int [pk, increment]
  transaction_id int
  fnb_item_id int
  quantity int
  sub_total float
}

Table payment_link {
  id int [pk, increment]
  transaction_id int
  url varchar
  expired_at datetime [default: `now()`]
  status Payment_Link_Status [default: 'pending']
  created_at datetime [default: `now()`]
  updated_at datetime [default: `now()`]
}

Table user {
  id int [pk, increment]
  full_name varchar
  email varchar
  username varchar
  password varchar
  created_at datetime [default: `now()`]
  updated_at datetime [default: `now()`]
}

Table user_refresh_token {
  id int [pk, increment]
  user_id int
  token varchar
  expires_at datetime
  is_revoked boolean
  user_agent varchar
  ip_address varchar
  created_at datetime [default: `now()`]
}

// Foreign Key Relationships
Ref: transaction_item_unit.transaction_id > transaction.id [delete: cascade]
Ref: transaction_item_unit.unit_item_id > unit_item.id [delete: cascade]
Ref: transaction_item_fnb.transaction_id > transaction.id [delete: cascade]
Ref: transaction_item_fnb.fnb_item_id > fnb_item.id [delete: cascade]
Ref: payment_link.transaction_id > transaction.id [delete: cascade]
Ref: user_refresh_token.user_id > user.id [delete: cascade]
